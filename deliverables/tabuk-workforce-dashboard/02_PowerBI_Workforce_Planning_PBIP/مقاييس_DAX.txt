إجمالي الموظفين =
COUNTROWS(Workforce)
Format: #,##0

على رأس العمل =
CALCULATE([إجمالي الموظفين], Workforce[الحالة_الوظيفية] = "على رأس العمل")
Format: #,##0

نسبة السعودة =
DIVIDE(CALCULATE([إجمالي الموظفين], Workforce[الجنسية] = "سعودي"), [إجمالي الموظفين])
Format: 0.0%

متوسط الأداء الصحيح =
AVERAGE(Workforce[تقييم_الأداء_2025])
Format: 0.0

رواتب على رأس العمل (ر.س) =
CALCULATE(SUM(Workforce[الراتب_الأساسي]), Workforce[الحالة_الوظيفية] = "على رأس العمل")
Format: #,##0

تصنيف منتهي - على رأس العمل =
CALCULATE([إجمالي الموظفين], Workforce[الحالة_الوظيفية] = "على رأس العمل", Workforce[حالة_التصنيف] = "منتهي")
Format: #,##0

تصنيفات منتهية - جميع الحالات =
CALCULATE([إجمالي الموظفين], Workforce[حالة_التصنيف] = "منتهي")
Format: #,##0

ينتهي خلال 90 يوم =
CALCULATE([إجمالي الموظفين], Workforce[حالة_التصنيف] = "ينتهي خلال 90 يوم")
Format: #,##0

حالة وظيفية مفقودة =
CALCULATE([إجمالي الموظفين], Workforce[الحالة_الوظيفية] = "غير محدد")
Format: #,##0

تعارض وظيفي محتمل =
CALCULATE([إجمالي الموظفين], Workforce[تعارض_الدور] = "يحتاج مراجعة")
Format: #,##0

رواتب صفرية =
CALCULATE([إجمالي الموظفين], Workforce[الراتب_الأساسي] = 0)
Format: #,##0

موظفون بملاحظات جودة =
CALCULATE([إجمالي الموظفين], Workforce[عدد_مشكلات_الجودة] > 0)
Format: #,##0

إجمالي ملاحظات الجودة =
SUM(Workforce[عدد_مشكلات_الجودة])
Format: #,##0
